# Everything except Depth Anything 3 itself, which has to go in separately:
#
#   pip install -r requirements.txt
#   pip install --no-deps depth-anything-3
#
# Its own dependency list pins a different Torch and pulls in a reconstruction
# and visualisation stack the app never calls, so it is installed without one.
# Everything it actually reaches for at runtime is below, each checked by
# blocking the import and converting a photo.
#
# `evo` is deliberately absent.  DA3 imports it to align camera poses, which a
# monocular app never does, and it is GPL-3.0 -- the one copyleft dependency in
# the tree.  `depth.DepthEstimator._load_da3` stubs the module that reaches for
# it, which keeps the whole distribution permissively licensed.
torch>=2.1
torchvision
einops
addict
omegaconf
tqdm
imageio
opencv-python
pillow>=10.0
pillow-heif>=0.16
numpy>=1.24,<2

# transformers and huggingface_hub are a pair, and the pin is load-bearing.
# transformers 4.57 declares hub <1.0 and refuses to import above it, so a tree
# that ends up with 4.57 and hub 1.x has a working `da3` and a *broken* `da2-*`
# -- the fallback models import transformers and the older one will not have it.
# That was true here before anything was pinned, and is the kind of breakage that
# only shows up when someone reaches for the fallback.  The modern pair works for
# both, checked by converting a photo with each.
transformers>=5.0
huggingface_hub>=1.0
